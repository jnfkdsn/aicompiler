cmake_minimum_required(VERSION 3.20)
project(mlir_conversion_patterns LANGUAGES C CXX)
find_package(MLIR REQUIRED CONFIG)
list(APPEND CMAKE_MODULE_PATH "${MLIR_CMAKE_DIR}" "${LLVM_CMAKE_DIR}")
include(TableGen)
include(AddLLVM)
include(AddMLIR)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include_directories(SYSTEM ${LLVM_INCLUDE_DIRS} ${MLIR_INCLUDE_DIRS})
include_directories(${CMAKE_CURRENT_SOURCE_DIR} ${CMAKE_CURRENT_BINARY_DIR})
add_definitions(${LLVM_DEFINITIONS})
if(NOT LLVM_ENABLE_RTTI)
  add_compile_options(-fno-rtti)
endif()
set(LLVM_TARGET_DEFINITIONS MyOps.td)
mlir_tablegen(MyDialect.h.inc -gen-dialect-decls -dialect=my)
mlir_tablegen(MyDialect.cpp.inc -gen-dialect-defs -dialect=my)
mlir_tablegen(MyOps.h.inc -gen-op-decls)
mlir_tablegen(MyOps.cpp.inc -gen-op-defs)
add_public_tablegen_target(MyExtensionIncGen)
set(LLVM_TARGET_DEFINITIONS Patterns.td)
mlir_tablegen(MyPatterns.inc -gen-rewriters)
add_public_tablegen_target(MyPatternsGen)
add_library(MyIR STATIC MyOps.cpp)
add_dependencies(MyIR MyExtensionIncGen)
target_link_libraries(MyIR PUBLIC MLIRIR MLIRSideEffectInterfaces)
add_executable(my-opt my-opt.cpp Flatten.cpp Declarative.cpp)
add_dependencies(my-opt MyPatternsGen)
target_link_libraries(my-opt PRIVATE MyIR MLIROptLib MLIRArithDialect
 MLIRControlFlowDialect MLIRFuncDialect MLIRFuncTransforms MLIRTransforms
 MLIRPDLDialect MLIRPDLInterpDialect MLIRRewritePDL)
