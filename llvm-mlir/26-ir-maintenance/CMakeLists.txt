cmake_minimum_required(VERSION 3.20)
project(mlir_ir_maintenance LANGUAGES C CXX)
find_package(MLIR REQUIRED CONFIG)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_definitions(${LLVM_DEFINITIONS})
if(NOT LLVM_ENABLE_RTTI)
 add_compile_options(-fno-rtti)
endif()
if(LLVM_ENABLE_ASSERTIONS)
 add_compile_options(-UNDEBUG)
endif()
add_executable(inspect inspect.cpp)
target_include_directories(inspect SYSTEM PRIVATE ${LLVM_INCLUDE_DIRS} ${MLIR_INCLUDE_DIRS})
target_link_libraries(inspect PRIVATE MLIRParser MLIRIR MLIRSupport MLIRArithDialect
 MLIRControlFlowDialect MLIRFuncDialect MLIRMemRefDialect MLIRSCFDialect MLIRSideEffectInterfaces)
