cmake_minimum_required(VERSION 3.20)
project(mlir_ods_storage LANGUAGES C CXX)
find_package(MLIR REQUIRED CONFIG)
list(APPEND CMAKE_MODULE_PATH "${MLIR_CMAKE_DIR}" "${LLVM_CMAKE_DIR}")
include(TableGen)
include(AddLLVM)
include(AddMLIR)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include_directories(SYSTEM ${LLVM_INCLUDE_DIRS} ${MLIR_INCLUDE_DIRS})
include_directories(${CMAKE_CURRENT_SOURCE_DIR} ${CMAKE_CURRENT_BINARY_DIR})
add_definitions(${LLVM_DEFINITIONS})
if(NOT LLVM_ENABLE_RTTI)
 add_compile_options(-fno-rtti)
endif()
set(LLVM_TARGET_DEFINITIONS MyOps.td)
mlir_tablegen(MyDialect.h.inc -gen-dialect-decls -dialect=my)
mlir_tablegen(MyDialect.cpp.inc -gen-dialect-defs -dialect=my)
mlir_tablegen(MyTypes.h.inc -gen-typedef-decls -typedefs-dialect=my)
mlir_tablegen(MyTypes.cpp.inc -gen-typedef-defs -typedefs-dialect=my)
mlir_tablegen(MyAttrs.h.inc -gen-attrdef-decls -attrdefs-dialect=my)
mlir_tablegen(MyAttrs.cpp.inc -gen-attrdef-defs -attrdefs-dialect=my)
mlir_tablegen(MyTypeInterfaces.h.inc -gen-type-interface-decls)
mlir_tablegen(MyAttrInterfaces.h.inc -gen-attr-interface-decls)
mlir_tablegen(MyTypeInterfaces.cpp.inc -gen-type-interface-defs)
mlir_tablegen(MyAttrInterfaces.cpp.inc -gen-attr-interface-defs)
mlir_tablegen(MyOps.h.inc -gen-op-decls)
mlir_tablegen(MyOps.cpp.inc -gen-op-defs)
add_public_tablegen_target(MyIncGen)
add_library(MyIR STATIC MyOps.cpp)
add_dependencies(MyIR MyIncGen)
target_link_libraries(MyIR PUBLIC MLIRIR MLIRSideEffectInterfaces)
add_executable(my-opt my-opt.cpp Segments.cpp)
target_link_libraries(my-opt PRIVATE MyIR MLIROptLib MLIRFuncDialect MLIRArithDialect MLIRPass MLIRTransforms)
add_executable(storage-probe storage-probe.cpp)
target_link_libraries(storage-probe PRIVATE MyIR)
