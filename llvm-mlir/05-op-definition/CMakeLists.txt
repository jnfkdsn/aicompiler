cmake_minimum_required(VERSION 3.20)
project(mlir_op_definition LANGUAGES C CXX)
find_package(MLIR REQUIRED CONFIG)
list(APPEND CMAKE_MODULE_PATH "${MLIR_CMAKE_DIR}" "${LLVM_CMAKE_DIR}")
include(TableGen)
include(AddLLVM)
include(AddMLIR)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include_directories(SYSTEM ${LLVM_INCLUDE_DIRS} ${MLIR_INCLUDE_DIRS})
include_directories(${CMAKE_CURRENT_SOURCE_DIR} ${CMAKE_CURRENT_BINARY_DIR})
add_definitions(${LLVM_DEFINITIONS})
if(NOT LLVM_ENABLE_RTTI)
  add_compile_options(-fno-rtti)
endif()
# BEGIN: tablegen
set(LLVM_TARGET_DEFINITIONS MyOps.td)
mlir_tablegen(MyOps.h.inc -gen-op-decls)
mlir_tablegen(MyOps.cpp.inc -gen-op-defs)
mlir_tablegen(MyDialect.h.inc -gen-dialect-decls -dialect=my)
mlir_tablegen(MyDialect.cpp.inc -gen-dialect-defs -dialect=my)
add_public_tablegen_target(MyOpsIncGen)
# END: tablegen
add_library(MyOps STATIC MyOps.cpp)
add_dependencies(MyOps MyOpsIncGen)
target_link_libraries(MyOps PUBLIC MLIRIR MLIRSideEffectInterfaces)
add_executable(my-opt my-opt.cpp ExpandClamp.cpp)
target_link_libraries(my-opt PRIVATE MyOps MLIROptLib MLIRFuncDialect
  MLIRArithDialect MLIRPass MLIRTransforms MLIRRewrite)
add_executable(my-build my-build.cpp)
target_link_libraries(my-build PRIVATE MyOps MLIRFuncDialect)
