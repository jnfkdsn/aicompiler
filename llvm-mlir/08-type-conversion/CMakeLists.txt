cmake_minimum_required(VERSION 3.20)
project(mlir_range_conversion LANGUAGES C CXX)
find_package(MLIR REQUIRED CONFIG)
list(APPEND CMAKE_MODULE_PATH "${MLIR_CMAKE_DIR}" "${LLVM_CMAKE_DIR}")
include(TableGen)
include(AddLLVM)
include(AddMLIR)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include_directories(SYSTEM ${LLVM_INCLUDE_DIRS} ${MLIR_INCLUDE_DIRS})
include_directories(${CMAKE_CURRENT_SOURCE_DIR} ${CMAKE_CURRENT_SOURCE_DIR}/../06-ir-definition
                    ${CMAKE_CURRENT_BINARY_DIR}/reference)
add_definitions(${LLVM_DEFINITIONS})
if(NOT LLVM_ENABLE_RTTI)
  add_compile_options(-fno-rtti)
endif()
add_subdirectory(../06-ir-definition reference EXCLUDE_FROM_ALL)
add_library(MyRangeConversion STATIC RangeConversion.cpp)
target_link_libraries(MyRangeConversion PUBLIC MyIR MLIRArithDialect MLIRControlFlowDialect
  MLIRFuncDialect MLIRFuncTransforms MLIRPass MLIRTransformUtils)
add_executable(range-opt range-opt.cpp)
target_link_libraries(range-opt PRIVATE MyRangeConversion MLIROptLib MLIRTransforms MLIRReconcileUnrealizedCasts)
add_executable(range-probe range-probe.cpp)
target_link_libraries(range-probe PRIVATE MyRangeConversion MLIRParser)
