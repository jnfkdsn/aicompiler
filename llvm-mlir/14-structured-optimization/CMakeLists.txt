cmake_minimum_required(VERSION 3.20)
project(mlir_structured_analysis LANGUAGES C CXX)
find_package(MLIR REQUIRED CONFIG)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(inspect-dependence dependence.cpp)
target_include_directories(inspect-dependence SYSTEM PRIVATE ${LLVM_INCLUDE_DIRS} ${MLIR_INCLUDE_DIRS})
add_definitions(${LLVM_DEFINITIONS})
if(NOT LLVM_ENABLE_RTTI)
  target_compile_options(inspect-dependence PRIVATE -fno-rtti)
endif()
target_link_libraries(inspect-dependence PRIVATE MLIRParser MLIRAffineAnalysis MLIRAffineUtils
  MLIRAffineDialect MLIRArithDialect MLIRFuncDialect MLIRMemRefDialect MLIRIR MLIRSupport)
