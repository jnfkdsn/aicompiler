cmake_minimum_required(VERSION 3.20)
project(mlir_bytecode_plugins LANGUAGES C CXX)
find_package(MLIR REQUIRED CONFIG)
list(APPEND CMAKE_MODULE_PATH "${MLIR_CMAKE_DIR}" "${LLVM_CMAKE_DIR}")
include(TableGen)
include(AddLLVM)
include(AddMLIR)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include_directories(SYSTEM ${LLVM_INCLUDE_DIRS} ${MLIR_INCLUDE_DIRS})
include_directories(${CMAKE_CURRENT_SOURCE_DIR} ${CMAKE_CURRENT_BINARY_DIR})
add_definitions(${LLVM_DEFINITIONS})
if(NOT LLVM_ENABLE_RTTI)
 add_compile_options(-fno-rtti)
endif()
if(LLVM_ENABLE_ASSERTIONS)
 add_compile_options(-UNDEBUG)
endif()
set(LLVM_TARGET_DEFINITIONS MyOps.td)
mlir_tablegen(MyDialect.h.inc -gen-dialect-decls -dialect=my)
mlir_tablegen(MyDialect.cpp.inc -gen-dialect-defs -dialect=my)
mlir_tablegen(MyOps.h.inc -gen-op-decls)
mlir_tablegen(MyOps.cpp.inc -gen-op-defs)
add_public_tablegen_target(MyVersionIncGen)
foreach(version 1 2)
 add_executable(my-opt-v${version} MyOps.cpp my-opt.cpp)
 add_dependencies(my-opt-v${version} MyVersionIncGen)
 target_compile_definitions(my-opt-v${version} PRIVATE MY_DIALECT_VERSION=${version})
 target_link_libraries(my-opt-v${version} PRIVATE MLIROptLib MLIRIR MLIRSideEffectInterfaces MLIRArithDialect MLIRFuncDialect MLIRTransforms)
endforeach()
# Linux module resolves MLIR symbols from this build's exported mlir-opt.
add_library(MyPlugin MODULE Plugin.cpp MyOps.cpp)
add_dependencies(MyPlugin MyVersionIncGen)
target_compile_definitions(MyPlugin PRIVATE MY_DIALECT_VERSION=2)
set_target_properties(MyPlugin PROPERTIES PREFIX "")
add_library(BadPlugin MODULE BadPlugin.cpp)
set_target_properties(BadPlugin PROPERTIES PREFIX "")
add_executable(plugin-check plugin-check.cpp)
target_link_libraries(plugin-check PRIVATE MLIRPluginsLib)
set_target_properties(plugin-check PROPERTIES ENABLE_EXPORTS ON)
